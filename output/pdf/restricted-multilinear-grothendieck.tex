\title[Persistence of the trilinear obstruction]
{Persistence of the trilinear operator space Grothendieck obstruction under complete retractions}

\author{Draft manuscript}
\address{}
\email{}

\date{\today}

\begin{document}

\begin{abstract}
Briet and Palazuelos proved that the trilinear operator space
Grothendieck inequality fails already on finite dimensional commutative
$C^*$-algebras. We record a permanence principle showing that this obstruction
passes through completely bounded retractions. Consequently the quantitative
failure persists, with the same polynomial order up to constants, on full
matrix algebras, on $B(H)$ for infinite dimensional $H$, and on every finite
von Neumann algebra containing arbitrarily large finite dimensional abelian
subalgebras with contractive conditional expectations. We also prove the
corresponding lower bound for weighted symmetrized Haagerup-type envelopes:
any uniformly bounded weighted modification must still fail on matrix
algebras; a uniform weighted theorem would force the smallest weight on the
commutative diagonal corner to decay polynomially. No conjectural statements
are used.
\end{abstract}

\section{Introduction}

The bilinear operator space Grothendieck theorem says that the jointly
completely bounded norm and the symmetrized completely bounded norm of a
bilinear form on $C^*$-algebras are equivalent with universal constant at most
$2$; this is the theorem of Pisier-Shlyakhtenko and Haagerup-Musat. Pisier
asked whether the analogous trilinear statement is true. Briet and Palazuelos
answered negatively: there are trilinear forms on $(\ell_\infty^n)^3$ for
which the ratio of the symmetrized completely bounded norm to the jointly
completely bounded norm grows at least like a positive power of $n$.

The negative result is sometimes read as a purely commutative obstruction.
This note shows that the obstruction is not confined to the commutative
category. The simple mechanism is complete retraction. If three operator
spaces contain completely complemented copies of the spaces on which the
trilinear obstruction occurs, then the obstruction extends to the larger
triple. Since the diagonal algebra $D_n\cong \ell_\infty^n$ is the range of a
completely positive contractive projection from $M_n$, the Briet-Palazuelos
forms extend from $D_n$ to $M_n$ without losing their asymptotic separation.
Thus full matrix algebras, although finite dimensional, nuclear, exact, and
simple, cannot support an unrestricted trilinear operator space
Grothendieck theorem.

The second point concerns modified symmetrized norms. A natural response to
the failure theorem is to replace the six-fold symmetrized norm by a weighted
version. We prove a necessary calibration condition for any such weighted
theory. If the smallest weight stays bounded below on diagonal corners, then
the same polynomial obstruction survives on matrix algebras. Hence a weighted
theorem uniform over matrix algebras must make at least one ordering
polynomially cheap on the commutative diagonal.

\section{Trilinear constants}

Let $E_1,E_2,E_3$ be operator spaces and let
\[
   u:E_1\times E_2\times E_3\longrightarrow \C
\]
be a trilinear form. The jointly completely bounded amplification is
\[
   u_d^{\jcb}\left(\sum_i x_i^1\ot a_i^1,
                  \sum_j x_j^2\ot a_j^2,
                  \sum_k x_k^3\ot a_k^3\right)
   =
   \sum_{i,j,k}u(x_i^1,x_j^2,x_k^3)\,
      a_i^1\ot a_j^2\ot a_k^3
\]
as an element of $M_d\ot M_d\ot M_d$. Put
\[
   \|u\|_{\jcb}
   =
   \sup_d \|u_d^{\jcb}\|.
\]

For an ordered trilinear form $v:E_1\times E_2\times E_3\to\C$ define
\[
   \|v\|_{\cb}
   =
   \sup_d
   \left\|
      \sum_{i,j,k}v(x_i^1,x_j^2,x_k^3)\,
      a_i^1 a_j^2 a_k^3
   \right\|,
\]
where the supremum is over contractions in $M_d(E_1)$, $M_d(E_2)$, and
$M_d(E_3)$. For $\pi\in S_3$ write
\[
   u^\pi(x_{\pi(1)},x_{\pi(2)},x_{\pi(3)})
   =
   u(x_1,x_2,x_3).
\]

\begin{definition}
The symmetrized completely bounded norm of $u$ is
\[
   \|u\|_{\sym}
   =
   \inf
   \left\{
      \sum_{\pi\in S_3}\|v_\pi^\pi\|_{\cb}:
      u=\sum_{\pi\in S_3}v_\pi
   \right\}.
\]
The trilinear operator space Grothendieck constant of the triple
$(E_1,E_2,E_3)$ is
\[
   \GT(E_1,E_2,E_3)
   =
   \sup_{u\ne0}\frac{\|u\|_{\sym}}{\|u\|_{\jcb}}.
\]
\end{definition}

The inequality $\|u\|_{\jcb}\le \|u\|_{\sym}$ follows from the standard
commuting representation trick: in the ordered amplification use
$a_i^1\ot I\ot I$, $I\ot a_j^2\ot I$, and $I\ot I\ot a_k^3$.

\begin{definition}
Let $\omega=(\omega_\pi)_{\pi\in S_3}$ be positive weights. The weighted
symmetrized norm is
\[
   \|u\|_{\sym,\omega}
   =
   \inf
   \left\{
      \sum_{\pi\in S_3}\omega_\pi\|v_\pi^\pi\|_{\cb}:
      u=\sum_{\pi\in S_3}v_\pi
   \right\}.
\]
Set
\[
   \GT_\omega(E_1,E_2,E_3)
   =
   \sup_{u\ne0}\frac{\|u\|_{\sym,\omega}}{\|u\|_{\jcb}},
   \qquad
   \minw(\omega)=\min_{\pi\in S_3}\omega_\pi .
\]
\end{definition}

Since
\[
   \|u\|_{\sym,\omega}\ge \minw(\omega)\|u\|_{\sym},
\]
all lower bounds for $\GT$ immediately give weighted lower bounds after
multiplication by $\minw(\omega)$.

\section{Complete retractions}

\begin{definition}
Let $F$ and $E$ be operator spaces. We say that $F$ is completely
$\lambda$-retracted in $E$ if there are completely bounded maps
\[
   j:F\to E,\qquad P:E\to F
\]
such that $Pj=\id_F$ and
\[
   \|j\|_{\cb}\|P\|_{\cb}\le \lambda .
\]
If $\lambda=1$, we say completely contractively retracted.
\end{definition}

\begin{theorem}[Retraction monotonicity]
\label{thm:retraction}
Let $F_i$ be completely $\lambda_i$-retracted in $E_i$ for $i=1,2,3$. Then
\[
   \GT(E_1,E_2,E_3)
   \ge
   \frac{\GT(F_1,F_2,F_3)}{\lambda_1\lambda_2\lambda_3}.
\]
More generally, for every positive weight vector $\omega$,
\[
   \GT_\omega(E_1,E_2,E_3)
   \ge
   \frac{\GT_\omega(F_1,F_2,F_3)}{\lambda_1\lambda_2\lambda_3}.
\]
\end{theorem}

\begin{proof}
Choose maps $j_i:F_i\to E_i$ and $P_i:E_i\to F_i$ with $P_i j_i=\id_{F_i}$.
Let $u:F_1\times F_2\times F_3\to\C$ be a trilinear form and define its
extension
\[
   \widetilde u(x_1,x_2,x_3)
   =
   u(P_1x_1,P_2x_2,P_3x_3).
\]
Functoriality of the jointly completely bounded amplification gives
\[
   \|\widetilde u\|_{\jcb}
   \le
   \|P_1\|_{\cb}\|P_2\|_{\cb}\|P_3\|_{\cb}\|u\|_{\jcb}.
\]
Since
\[
   u(y_1,y_2,y_3)
   =
   \widetilde u(j_1y_1,j_2y_2,j_3y_3),
\]
the same functoriality for each ordered completely bounded norm gives
\[
   \|u\|_{\sym,\omega}
   \le
   \|j_1\|_{\cb}\|j_2\|_{\cb}\|j_3\|_{\cb}
   \|\widetilde u\|_{\sym,\omega}.
\]
Therefore
\[
   \frac{\|\widetilde u\|_{\sym,\omega}}{\|\widetilde u\|_{\jcb}}
   \ge
   \frac{1}{\prod_i \|j_i\|_{\cb}\|P_i\|_{\cb}}
   \frac{\|u\|_{\sym,\omega}}{\|u\|_{\jcb}}.
\]
Taking the supremum over $u$ and then the infimum over the retraction
constants proves the weighted statement. The unweighted statement is the case
$\omega_\pi=1$.
\end{proof}

\section{Matrix algebras}

Let $D_n\subset M_n$ be the diagonal algebra. The inclusion
$D_n\hookrightarrow M_n$ is a complete isometry and the diagonal projection
\[
   \mathbb E_n((a_{ij})_{i,j=1}^n)=\diag(a_{11},\ldots,a_{nn})
\]
is a completely positive contraction. Hence $D_n$ is completely
contractively retracted in $M_n$.

We use the following quantitative theorem of Briet and Palazuelos.

\begin{theorem}[Briet-Palazuelos]
\label{thm:bp}
There are constants $a>0$ and $\eta>0$ such that for every $n\ge 2$ there is
a trilinear form $u_n:(\ell_\infty^n)^3\to\C$ satisfying
\[
   \|u_n\|_{\sym}\ge a n^\eta\|u_n\|_{\jcb}.
\]
In their construction one may take $\eta=1/8$ up to absolute changes in the
constant.
\end{theorem}

\begin{theorem}[Failure on full matrix algebras]
\label{thm:matrix}
There are constants $a>0$ and $\eta>0$ such that, for every $n\ge2$,
\[
   \GT(M_n,M_n,M_n)\ge a n^\eta .
\]
Moreover, for every positive weight vector $\omega$,
\[
   \GT_\omega(M_n,M_n,M_n)\ge a\,\minw(\omega)\,n^\eta .
\]
\end{theorem}

\begin{proof}
Identify $\ell_\infty^n$ with $D_n$. By Theorem~\ref{thm:bp},
\[
   \GT(D_n,D_n,D_n)\ge a n^\eta .
\]
Since $D_n$ is completely contractively retracted in $M_n$,
Theorem~\ref{thm:retraction} gives the unweighted estimate. For the weighted
estimate, use
\[
   \GT_\omega(D_n,D_n,D_n)
   \ge
   \minw(\omega)\GT(D_n,D_n,D_n)
\]
and apply Theorem~\ref{thm:retraction} again.
\end{proof}

\begin{corollary}
\label{cor:no-exact-nuclear}
No trilinear operator space Grothendieck inequality with a constant depending
only on exactness, nuclearity, or finite dimensionality can hold for
$C^*$-algebras. In particular, the obstruction persists on the simple nuclear
exact algebras $M_n$ with polynomial growth in $n$.
\end{corollary}

\begin{proof}
Each $M_n$ is finite dimensional, nuclear, exact, and simple. A bound
depending only on any of these qualitative properties would give a uniform
upper bound for $\GT(M_n,M_n,M_n)$, contradicting
Theorem~\ref{thm:matrix}.
\end{proof}

\begin{corollary}[Calibration of weighted modifications]
\label{cor:weights}
Let $\omega^{(n)}$ be positive weights assigned to triples
$(M_n,M_n,M_n)$. If
\[
   \sup_{n\ge2}\GT_{\omega^{(n)}}(M_n,M_n,M_n)<\infty ,
\]
then
\[
   \minw(\omega^{(n)})=O(n^{-\eta})
\]
with $\eta$ as in Theorem~\ref{thm:bp}. In particular, every weighted theory
whose weights are uniformly bounded below fails uniformly on matrix
algebras.
\end{corollary}

\begin{proof}
This follows immediately from the weighted part of Theorem~\ref{thm:matrix}.
\end{proof}

\section{Infinite dimensional consequences}

The finite dimensional estimates imply unbounded trilinear constants for many
standard infinite dimensional operator algebras.

\begin{corollary}[The algebra \(B(H)\)]
\label{cor:bh}
If $H$ is infinite dimensional, then
\[
   \GT(B(H),B(H),B(H))=\infty .
\]
The same conclusion holds for $\GT_\omega$ whenever the weights are bounded
below by a positive constant.
\end{corollary}

\begin{proof}
For every $n$, choose orthonormal vectors $e_1,\ldots,e_n$ in $H$ and let
$D_n$ be the diagonal algebra supported on their span. The map
\[
   T\mapsto \sum_{k=1}^n \langle Te_k,e_k\rangle e_k\ot e_k^*
\]
is completely positive and contractive onto $D_n$, and the inclusion of
$D_n$ into $B(H)$ is completely isometric. Thus $D_n$ is completely
contractively retracted in $B(H)$ for every $n$. Apply
Theorems~\ref{thm:retraction} and~\ref{thm:bp} and let $n\to\infty$.
\end{proof}

\begin{corollary}[Finite von Neumann algebras]
\label{cor:finite-vn}
Let $M$ be a finite von Neumann algebra with a faithful normal trace. Suppose
that for arbitrarily large $n$ the algebra $M$ contains a unital abelian
subalgebra generated by $n$ nonzero orthogonal projections. Then
\[
   \GT(M,M,M)=\infty .
\]
In particular this holds for diffuse finite von Neumann algebras and for
type $\mathrm{II}_1$ factors.
\end{corollary}

\begin{proof}
In a finite von Neumann algebra, the trace-preserving conditional expectation
onto a von Neumann subalgebra is completely positive and contractive. Hence
each finite dimensional abelian subalgebra
$D_n\cong\ell_\infty^n$ appearing in the hypothesis is completely
contractively retracted in $M$. The conclusion follows from
Theorems~\ref{thm:retraction} and~\ref{thm:bp}.
\end{proof}

\section{Conclusion}

The trilinear obstruction is stable under completely bounded retractions.
Therefore exactness, nuclearity, noncommutativity, and passage from diagonal
algebras to full matrix algebras do not repair the failed trilinear operator
space Grothendieck theorem. Any viable replacement must exclude large
completely complemented commutative corners or weaken the symmetrized
Haagerup envelope by weights that degenerate at least polynomially on such
corners.
